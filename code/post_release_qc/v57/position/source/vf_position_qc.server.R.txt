#!/usr/bin/env Rscript
# VF 注释在 contig 上的位置:是 VF 注释的性质,还是承载它的 contig 的性质?
# 方法移植自 pc028e3 的 102-arg-position-qc(ARG 0.064× / AMG 1.084×),
# 把每条注释与「同 contig 长度、同基因密度」层内抽出的背景基因比较末端距离。
# AMG 作阴性对照:同类数据库命中,落在装配边界上没有先验理由。
# 只读 analyses/p0061.sqlite(canonical 3,060 vOTU),产出 dev/210-vf-position-qc/。
suppressPackageStartupMessages({
  library(data.table); library(DBI); library(RSQLite)
})

root <- "/home/allen/github/rujinlong/p0061-RecWaterVirome"
out  <- file.path(root, "dev/210-vf-position-qc")
dir.create(out, showWarnings = FALSE, recursive = TRUE)

con <- dbConnect(SQLite(), file.path(root, "analyses/p0061.sqlite"))
genes <- as.data.table(dbGetQuery(con, "
  SELECT vcontig_id, protein_id, CAST(gene_position AS INTEGER) AS gene_ordinal,
         CAST(start_position AS INTEGER) AS gene_start,
         CAST(end_position   AS INTEGER) AS gene_end, ko_id
  FROM virus_gene_annotation_dramv"))
ann <- as.data.table(dbReadTable(con, "virus_contig_annotation"))[
  , .(vcontig_id, vOTU_id, contig_length, checkv_provirus, checkv_completeness,
      checkv_contamination, genomad_topology)]
vf  <- as.data.table(dbReadTable(con, "virus_vfdbA_filtered"))[, .(vOTU_id, vcontig_id, query, vf_function, pident)]
amg <- as.data.table(dbReadTable(con, "virus_amg_dramv"))[, .(vcontig_id, orf_id)]
dbDisconnect(con)

# ---- 基因坐标与分层 -------------------------------------------------------
genes <- ann[genes, on = "vcontig_id", nomatch = 0L]
# ⚠ 带末端重复的完整基因组(DTR/ITR)其"末端"是真实基因组端点,不是装配断点,
#   末端距离在它们身上没有装配学含义 —— 整个检验只在线性 contig 上做。
n_circ <- uniqueN(genes[genomad_topology %in% c("DTR", "ITR"), vcontig_id])
genes <- genes[!genomad_topology %in% c("DTR", "ITR")]
genes[, d_end := pmin(gene_start - 1L, contig_length - gene_end)]
genes <- genes[!is.na(d_end) & d_end >= 0]
genes[, genes_on_contig := max(gene_ordinal), by = vcontig_id]
genes[, gene_density := genes_on_contig / (contig_length / 1000)]

len_breaks <- quantile(genes$contig_length, probs = seq(0, 1, 0.2), na.rm = TRUE)
den_breaks <- quantile(genes$gene_density, probs = c(0, 1/3, 2/3, 1), na.rm = TRUE)
assign_stratum <- function(dt) {
  dt[, stratum := paste(
    cut(contig_length, unique(len_breaks), include.lowest = TRUE, labels = FALSE),
    cut(gene_density, unique(den_breaks), include.lowest = TRUE, labels = FALSE),
    sep = "-")][]
}
assign_stratum(genes)

# ---- 注释集合 -------------------------------------------------------------
# VF query 与 AMG orf_id 同格式:vctg0998-cat_1_8 → protein_id vctg0998_8
orf_to_protein <- function(x) sub("^(vctg[0-9]+)-cat_[0-9]+_([0-9]+)$", "\\1_\\2", x)
vf[,  protein_id := orf_to_protein(query)]
amg[, protein_id := orf_to_protein(orf_id)]
vf_pos  <- genes[unique(vf[,  .(vcontig_id, protein_id)]), on = c("vcontig_id", "protein_id"), nomatch = 0L]
amg_pos <- genes[unique(amg[, .(vcontig_id, protein_id)]), on = c("vcontig_id", "protein_id"), nomatch = 0L]
# KO 注释作第二个阴性对照(数量级更大,层内抽样更稳)
ko_pos <- genes[!is.na(ko_id) & ko_id != ""]

cat(sprintf("排除带末端重复(DTR/ITR)的完整基因组 %d 条\n", n_circ))
cat(sprintf("背景基因 %s 条 / 线性 contig %s 条\n", format(nrow(genes), big.mark = ","),
            format(uniqueN(genes$vcontig_id), big.mark = ",")))
cat(sprintf("VF 解析到坐标 %d / %d ;AMG(DRAM-v) %d ;KO 注释 %s\n",
            nrow(vf_pos), nrow(unique(vf[, .(vcontig_id, protein_id)])),
            nrow(amg_pos), format(nrow(ko_pos), big.mark = ",")))

# ---- 层匹配 bootstrap 比值 -------------------------------------------------
ratio_vs_background <- function(focal, background, n_boot = 2000L, seed = 1L) {
  w <- focal[, .N, by = stratum]
  pool <- split(background$d_end, background$stratum)
  pool <- pool[names(pool) %in% w$stratum]
  w <- w[stratum %in% names(pool)]
  f <- focal[stratum %in% w$stratum]
  set.seed(seed)
  draw_bg <- function() unlist(Map(function(s, n) sample(pool[[s]], n, replace = TRUE),
                                   w$stratum, w$N), use.names = FALSE)
  obs <- median(f$d_end) / median(draw_bg())
  boot <- vapply(seq_len(n_boot), function(i) {
    median(f[sample(.N, .N, replace = TRUE)]$d_end) / median(draw_bg())
  }, numeric(1))
  list(ratio = obs, lo = quantile(boot, .025, na.rm = TRUE),
       hi = quantile(boot, .975, na.rm = TRUE), n = nrow(f),
       med = median(f$d_end))
}

sets <- list(`VF (VFDB-A filtered)` = vf_pos,
             `AMG (DRAM-v, negative control)` = amg_pos,
             `KO-annotated genes (negative control)` = ko_pos)
res <- rbindlist(lapply(names(sets), function(nm) {
  r <- ratio_vs_background(sets[[nm]], genes)
  data.table(annotation_set = nm, n = r$n, median_d_end_bp = r$med,
             ratio_vs_background = round(r$ratio, 3),
             ci_lo = round(r$lo, 3), ci_hi = round(r$hi, 3))
}))
fwrite(res, file.path(out, "position_ratio.tsv"), sep = "\t")
print(res)

# ---- VF 命中按 CheckV 区段归属拆开 ----------------------------------------
# 只有 19 条 provirus-flagged contig 有 region 预测;其余没有归属信息。
vf_pos[, flagged := checkv_provirus %in% c("Yes", "TRUE", "True", 1, "1")]
by_flag <- vf_pos[, .(n = .N, median_d_end = median(d_end),
                      median_contig_len = median(contig_length)), by = flagged]
fwrite(by_flag, file.path(out, "vf_by_provirus_flag.tsv"), sep = "\t")
print(by_flag)

# ---- 完整基因组上的 VF 命中 ------------------------------------------------
# DTR/ITR 末端重复标志完整病毒基因组,宿主序列混入在结构上说不通。
# 这些命中被上面的线性检验排除,单独计数。
vf_circ <- ann[genomad_topology %in% c("DTR", "ITR")][vf, on = c("vcontig_id", "vOTU_id"),
                                                      nomatch = 0L]
circ_fun <- vf_circ[, .(n_hits = .N), by = vf_function][order(-n_hits)]
fwrite(circ_fun, file.path(out, "vf_on_complete_genomes.tsv"), sep = "\t")
cat(sprintf("\n--- 完整基因组(DTR/ITR)上的 VF 命中:%d 条 / %d 个 vOTU,pident 中位 %.1f%% ---\n",
            nrow(vf_circ), uniqueN(vf_circ$vOTU_id), median(vf_circ$pident)))
print(circ_fun)

# ---- host 区段长度:LT packaging gradient 的可测性 --------------------------
# LT 的 cargo 延伸到 attB 下游数十至上百 kb(Kuang 2026 在 E. coli 用 attB+120 kb 标记)。
host_len <- ann[checkv_provirus == "Yes",
                .(host_bp = round(contig_length * checkv_contamination / 100))]
cat("\n--- CheckV flagged contig 的 host 区段长度(bp;LT 可测性)---\n")
print(summary(host_len$host_bp))
cat(sprintf("host 区段 >20 kb 的条数:%d / %d\n",
            sum(host_len$host_bp > 20000), nrow(host_len)))

writeLines(capture.output(sessionInfo()), file.path(out, "session-info.txt"))
cat("\n产物写入:", out, "\n")

# ---- 图:三个注释集的末端距离比值(lollipop + 等效区间)---------------------
suppressPackageStartupMessages({ library(ggplot2) })
dt_p <- copy(res)
dt_p[, set := factor(annotation_set, levels = rev(annotation_set))]
p <- ggplot(dt_p, aes(x = ratio_vs_background, y = set)) +
  annotate("rect", xmin = 0.67, xmax = 1.5, ymin = -Inf, ymax = Inf,
           fill = "grey90", colour = NA) +
  geom_vline(xintercept = 1, linetype = 2, colour = "grey40") +
  geom_errorbar(aes(xmin = ci_lo, xmax = ci_hi), width = 0.12, colour = "grey25",
                orientation = "y") +
  geom_point(size = 3.4, colour = "#1b6ca8") +
  geom_text(aes(label = sprintf("%.2f [%.2f, %.2f]  n=%d", ratio_vs_background,
                                ci_lo, ci_hi, n)), vjust = -1.25, size = 3.1) +
  scale_x_continuous(trans = "log10", breaks = c(0.5, 0.67, 1, 1.5, 2)) +
  coord_cartesian(xlim = c(0.55, 1.9)) +
  labs(x = "Median distance to contig end, relative to stratum-matched background (log scale)",
       y = NULL,
       title = "VF annotations sit where background genes sit",
       subtitle = "Grey band: equivalence bounds 0.67-1.5. Dashed line: no positional difference.") +
  theme_bw(base_size = 11) +
  theme(panel.grid.minor = element_blank(), plot.title.position = "plot")
ggsave(file.path(out, "fig210a-vf-position-ratio.png"), p,
       width = 8.2, height = 2.9, dpi = 300)
cat("图写入: fig210a-vf-position-ratio.png\n")
