from pathlib import Path
from lxml import etree as E
import csv,hashlib,json,zipfile
P=Path(__file__).resolve().parents[2]
src=P/'07_Code/workflow/analyses/data/320-data1-local/Data1_local.xlsx'
dest=P/'05_Supplementary_Data/Data1_local.xlsx'
NS={'s':'http://schemas.openxmlformats.org/spreadsheetml/2006/main','r':'http://schemas.openxmlformats.org/officeDocument/2006/relationships'}
T='{'+NS['s']+'}'
with zipfile.ZipFile(src) as z:infos=z.infolist();parts={n:z.read(n) for n in z.namelist()}
parts['docProps/core.xml']=(Path(__file__).resolve().parent/'publication-core.xml').read_bytes()
w=E.fromstring(parts['xl/workbook.xml']);rels={x.get('Id'):x.get('Target') for x in E.fromstring(parts['xl/_rels/workbook.xml.rels'])}
s=next(x for x in w.findall('s:sheets/s:sheet',NS) if x.get('name')=='S37_gaps');target=rels[s.get('{'+NS['r']+'}id')];target=target.lstrip('/') if target.startswith('/') else 'xl/'+target
r=E.fromstring(parts[target]);shared=E.fromstring(parts['xl/sharedStrings.xml']) if 'xl/sharedStrings.xml' in parts else None
def value(c):
 if c.get('t')=='s':return ''.join(shared[int(c.find('s:v',NS).text)].itertext())
 return ''.join(c.find('s:is',NS).itertext()) if c.get('t')=='inlineStr' else (c.findtext('s:v',namespaces=NS) or '')
changes={
 'B2':'Historical source-project gap: original unmasked curve values were not exported. Current retained-table rarefaction values are supplied in Data 1 S50 and displayed in Supplementary Figure 1c.',
 'C2':'S50 supports the retained-table subsampling display with the full-depth detection mask fixed; it does not requalify breadth after raw-read subsampling or establish complete sampling of environmental viruses.'}
audit=[]
for address,new in changes.items():
 c=next(x for x in r.findall('.//s:c',NS) if x.get('r')==address);old=value(c);audit.append({'cell':address,'before':old,'after':new})
 for el in list(c):c.remove(el)
 c.set('t','inlineStr');E.SubElement(E.SubElement(c,T+'is'),T+'t').text=new
parts[target]=E.tostring(r,encoding='UTF-8',xml_declaration=True)
with zipfile.ZipFile(dest,'w',zipfile.ZIP_DEFLATED) as z:
 for info in infos:z.writestr(info,parts[info.filename])
rows=[]
for row in r.findall('s:sheetData/s:row',NS):rows.append([value(c) for c in row.findall('s:c',NS)])
with (dest.parent/'Data1_S37_gaps.tsv').open('w') as f:csv.writer(f,delimiter='\t',lineterminator='\n').writerows(rows)
(Path(__file__).resolve().parent/'metadata-patch.json').write_text(json.dumps({'sheet':'S37_gaps','changes':audit,'source_sha256':hashlib.sha256(src.read_bytes()).hexdigest(),'output_sha256':hashlib.sha256(dest.read_bytes()).hexdigest(),'scientific_values_changed':False},indent=2)+'\n')
print('Patched two S37 metadata cells; frozen workbook remains unchanged')
